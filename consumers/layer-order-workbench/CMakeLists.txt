cmake_minimum_required(VERSION 3.21)
project(LocalLayerOrderWorkbench LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(WORKBENCH_SANITIZERS "Instrument consumer and real Mini dependency" OFF)
if(NOT MINI_SOURCE)
    message(FATAL_ERROR "Pass -DMINI_SOURCE=<Mini v1.0.0 source directory>")
endif()
get_filename_component(MINI_SOURCE "${MINI_SOURCE}" ABSOLUTE)
if(NOT EXISTS "${MINI_SOURCE}/include/layerkeysort_mini.h")
    message(FATAL_ERROR "MINI_SOURCE must contain Mini's public header")
endif()
# An external project, not part of root Full or a modification of Mini.
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(LKS_MINI_STRICT ON CACHE BOOL "" FORCE)
set(LKS_MINI_SANITIZERS ${WORKBENCH_SANITIZERS} CACHE BOOL "" FORCE)
add_subdirectory("${MINI_SOURCE}" mini-dependency)
function(workbench_settings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
    if(WORKBENCH_SANITIZERS)
        if(WIN32 OR NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
            message(FATAL_ERROR "ASan/UBSan require a supported non-Windows runtime")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()
add_library(workbench_core STATIC src/command.c src/oracle.c src/workbench.c src/trace.c)
target_include_directories(workbench_core PUBLIC src)
target_link_libraries(workbench_core PUBLIC layerkeysort_mini)
workbench_settings(workbench_core)
add_executable(layer_order_workbench src/main.c)
target_link_libraries(layer_order_workbench PRIVATE workbench_core)
workbench_settings(layer_order_workbench)
option(WORKBENCH_TESTS "Enable the consumer's CTest suite" ON)
if(WORKBENCH_TESTS)
    enable_testing()
    find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
    add_executable(workbench_unit_tests tests/test_unit.c)
    target_link_libraries(workbench_unit_tests PRIVATE workbench_core)
    workbench_settings(workbench_unit_tests)
    add_test(NAME workbench_units COMMAND workbench_unit_tests)
    if((CMAKE_SYSTEM_NAME STREQUAL "Linux" OR MINGW) AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        add_executable(workbench_oom_tests tests/test_oom.c)
        target_link_libraries(workbench_oom_tests PRIVATE workbench_core)
        target_link_options(workbench_oom_tests PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=free)
        workbench_settings(workbench_oom_tests)
        add_test(NAME workbench_oom COMMAND workbench_oom_tests)
    endif()
    foreach(case scenes errors replay sizes)
        add_test(NAME workbench_${case}
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/check.py"
                --exe $<TARGET_FILE:layer_order_workbench> --case ${case}
                --output "${CMAKE_CURRENT_BINARY_DIR}/evidence/${case}")
        set_tests_properties(workbench_${case} PROPERTIES TIMEOUT 240)
    endforeach()
endif()
