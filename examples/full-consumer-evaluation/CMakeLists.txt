cmake_minimum_required(VERSION 3.21)
project(FullConsumerEvaluation LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(FULL_SOURCE "" CACHE PATH "Unmodified Full source checkout")
set(FULL_AMALGAMATION "" CACHE PATH "Extracted official distribution directory")
option(EVAL_SANITIZE "Instrument evaluation and Full with available sanitizers" OFF)
if(EVAL_SANITIZE)
  if(MSVC)
    add_compile_options(/fsanitize=address /Zi)
    add_link_options(/INCREMENTAL:NO)
  else()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
if(FULL_AMALGAMATION)
  add_library(full_distribution STATIC "${FULL_AMALGAMATION}/layerkeysort.c")
  target_include_directories(full_distribution PUBLIC "${FULL_AMALGAMATION}")
  set(FULL_TARGET full_distribution)
elseif(FULL_SOURCE)
  set(LKS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LKS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LKS_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${FULL_SOURCE}" full EXCLUDE_FROM_ALL)
  set(FULL_TARGET LayerKeySort::layerkeysort)
else()
  find_package(LayerKeySort 4 CONFIG REQUIRED)
  set(FULL_TARGET LayerKeySort::layerkeysort)
endif()
function(evaluation target)
  target_link_libraries(${target} PRIVATE ${FULL_TARGET})
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /D_CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(WIN32)
    target_link_libraries(${target} PRIVATE psapi)
  endif()
endfunction()
add_executable(full_editor editor.c model.c)
evaluation(full_editor)
add_executable(full_contracts contracts.c)
evaluation(full_contracts)
add_executable(full_measure measure.c)
evaluation(full_measure)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU" AND NOT EVAL_SANITIZE)
  # Linker instrumentation is external to the library; never a consumer API.
  add_executable(full_alloc allocation.c measure.c)
  target_compile_definitions(full_alloc PRIVATE EVAL_ALLOCATIONS)
  evaluation(full_alloc)
  add_executable(full_failures allocation.c failures.c)
  evaluation(full_failures)
  foreach(t full_alloc full_failures)
    target_link_options(${t} PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
  endforeach()
endif()
enable_testing()
add_test(NAME contracts COMMAND full_contracts)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME editor_model COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/exercise.py" $<TARGET_FILE:full_editor> --steps 12000)
add_test(NAME large_document COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/exercise.py" $<TARGET_FILE:full_editor> --large)
add_test(NAME document_workflow COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/exercise.py" $<TARGET_FILE:full_editor> --application)
if(TARGET full_failures)
  add_test(NAME allocation_failures COMMAND full_failures)
endif()
