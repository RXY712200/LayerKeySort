cmake_minimum_required(VERSION 3.21)
project(MiniExternalConsumer LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT MINI_SOURCE)
    message(FATAL_ERROR "Pass -DMINI_SOURCE=<independent Mini source directory>")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${MINI_SOURCE}" mini-build)
add_executable(mini_consumer consumer.c)
target_link_libraries(mini_consumer PRIVATE layerkeysort_mini)
enable_testing()
add_test(NAME external_mini_consumer COMMAND mini_consumer)
# Optional repository coexistence only. Never required by standalone Mini.
if(FULL_SOURCE)
    set(LKS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LKS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LKS_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${FULL_SOURCE}" full-build)
    add_executable(coexistence coexistence.c)
    target_link_libraries(coexistence PRIVATE layerkeysort layerkeysort_mini)
    add_test(NAME full_mini_coexistence COMMAND coexistence)
endif()
