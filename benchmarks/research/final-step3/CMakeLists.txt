cmake_minimum_required(VERSION 3.21)
project(FullFinalAcceptance LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(FULL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." CACHE PATH "Immutable Full implementation under test")
add_subdirectory(../history-step2 history-harness)
add_executable(full_layout layout.c)
target_include_directories(full_layout PRIVATE "${FULL_ROOT}/src")
if(TARGET full_research_diagnostics)
  target_compile_definitions(full_layout PRIVATE LKS_ENABLE_ALLOC_DIAGNOSTICS)
  target_link_libraries(full_layout PRIVATE full_research_diagnostics)
else()
  target_link_libraries(full_layout PRIVATE LayerKeySort::layerkeysort)
endif()
enable_testing()
add_test(NAME integrated_layout_lifetime COMMAND full_layout)
