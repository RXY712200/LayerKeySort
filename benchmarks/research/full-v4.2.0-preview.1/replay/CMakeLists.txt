cmake_minimum_required(VERSION 3.21)
project(ist_v05 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${FULL_SOURCE}" OUTPUT_VARIABLE pin OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT pin STREQUAL EXPECTED_FULL)
  message(FATAL_ERROR "Wrong current Full pin")
endif()
include("${FULL_SOURCE}/cmake/ProductionSources.cmake")
list(TRANSFORM LKS_SOURCES PREPEND "${FULL_SOURCE}/")
set(OLD "${BASELINE_SOURCE}")
foreach(mode prod diag)
  add_library(full_library_${mode} STATIC ${LKS_SOURCES})
  target_include_directories(full_library_${mode} PUBLIC "${FULL_SOURCE}/include" PRIVATE "${FULL_SOURCE}/src")
  if(mode STREQUAL diag)
    target_compile_definitions(full_library_${mode} PRIVATE LKS_ENABLE_ALLOC_DIAGNOSTICS)
  endif()
  foreach(kind full list tree)
    add_executable(${kind}_${mode} "${OLD}/harness.c" "${OLD}/baseline.c")
    if(kind STREQUAL full)
      target_compile_definitions(${kind}_${mode} PRIVATE FULL=1 TREE=0)
      target_link_libraries(${kind}_${mode} PRIVATE full_library_${mode})
      target_include_directories(${kind}_${mode} PRIVATE "${FULL_SOURCE}/src")
    elseif(kind STREQUAL tree)
      target_compile_definitions(${kind}_${mode} PRIVATE FULL=0 TREE=1)
    else()
      target_compile_definitions(${kind}_${mode} PRIVATE FULL=0 TREE=0)
    endif()
    if(mode STREQUAL diag)
      target_compile_definitions(${kind}_${mode} PRIVATE DIAGNOSTIC LKS_ENABLE_ALLOC_DIAGNOSTICS)
    endif()
    if(WIN32)
      target_link_libraries(${kind}_${mode} PRIVATE psapi)
    endif()
    if(NOT MSVC)
      target_compile_options(${kind}_${mode} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endforeach()
endforeach()
